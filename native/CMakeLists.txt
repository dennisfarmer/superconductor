cmake_minimum_required(VERSION 3.30)
project(conductor_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUNGEE_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory(${BUNGEE_SOURCE} bungee EXCLUDE_FROM_ALL)
add_library(conductor_bungee SHARED bungee_bridge.cpp)
target_include_directories(conductor_bungee PRIVATE ${BUNGEE_SOURCE})
target_link_libraries(conductor_bungee PRIVATE bungee_library)
